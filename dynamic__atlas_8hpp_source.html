196 void reserve(glm::ivec2 capacity)
197 {
198 if (capacity.x < size().x && capacity.y < size().y) {
199 return;
200 }
201 else if (m_texture.size() == glm::ivec2{}) {
202 m_texture.allocate(capacity);
203 m_texture.clear({});
204 }
205 else {
206 const tr::texture old_tex{m_texture.allocate(capacity)};
207 m_texture.clear({});
208 m_texture.copy_region({}, old_tex, {{}, old_tex.size()});

217 template <typename... Args>
218 requires(std::constructible_from<Value, rectangle<u16>, Args...>)
219 void add(Key key, sub_bitmap bitmap, Args&&... args)
220 {
221 std::optional<glm::u16vec2> tl{m_entries.try_insert(std::move(key), bitmap.size(), size(), std::forward<Args>(args)...)};
222 if (!tl.has_value()) {
223 glm::u16vec2 new_size;
224 if (m_texture.size() == glm::ivec2{}) {
225 const glm::uvec2 usize{bitmap.size()};
226 new_size = {std::bit_ceil(usize.x + 1), std::bit_ceil(usize.y + 1)};
227 tl = m_entries.try_insert(std::move(key), usize, new_size, std::forward<Args>(args)...);
228 }
229 else {
230 new_size = size();
231 do {
232 new_size.y < new_size.x ? new_size.y *= 2 : new_size.x *= 2;
233 } while (
234 !(tl = m_entries.try_insert(std::move(key), bitmap.size(), new_size, std::forward<Args>(args)...)).has_value());
235 }
236 reserve(new_size);
237 }
238 m_texture.set_region(*tl, bitmap);
239 }
