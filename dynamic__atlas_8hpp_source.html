257 void reserve(glm::ivec2 capacity)
258 {
259 if (capacity.x < size().x && capacity.y < size().y) {
260 return;
261 }
262 else if (m_texture.size() == glm::ivec2{}) {
263 m_texture.allocate(capacity);
264 m_texture.clear({});
265 }
266 else {
267 const tr::texture old_tex{m_texture.allocate(capacity)};
268 m_texture.clear({});
269 m_texture.copy_region({}, old_tex, {{}, old_tex.size()});
270 }

278 template <typename... Args>
279 requires(std::constructible_from<Value, rectangle<u16>, Args...>)
280 void add(Key key, sub_bitmap bitmap, Args&&... args)
281 {
282 std::optional<glm::u16vec2> tl{m_entries.try_insert(std::move(key), bitmap.size(), size(), std::forward<Args>(args)...)};
283 if (!tl.has_value()) {
284 glm::u16vec2 new_size;
285 if (m_texture.size() == glm::ivec2{}) {
286 const glm::uvec2 usize{bitmap.size()};
287 new_size = {std::bit_ceil(usize.x + 1), std::bit_ceil(usize.y + 1)};
288 tl = m_entries.try_insert(std::move(key), usize, new_size, std::forward<Args>(args)...);
289 }
290 else {
291 new_size = size();
292 do {
293 new_size.y < new_size.x ? new_size.y *= 2 : new_size.x *= 2;
294 } while (
295 !(tl = m_entries.try_insert(std::move(key), bitmap.size(), new_size, std::forward<Args>(args)...)).has_value());
296 }
297 reserve(new_size);
298 }
299 m_texture.set_region(*tl, bitmap);
300 }
