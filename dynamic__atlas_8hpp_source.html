237 void reserve(glm::ivec2 capacity)
238 {
239 if (capacity.x < size().x && capacity.y < size().y) {
240 return;
241 }
242 else if (m_texture.size() == glm::ivec2{}) {
243 m_texture.allocate(capacity);
244 m_texture.clear({});
245 }
246 else {
247 const tr::texture old_tex{m_texture.allocate(capacity)};
248 m_texture.clear({});
249 m_texture.copy_region({}, old_tex, {{}, old_tex.size()});
250 }

258 template <typename... Args>
259 requires(std::constructible_from<Value, rectangle<u16>, Args...>)
260 void add(Key key, sub_bitmap bitmap, Args&&... args)
261 {
262 std::optional<glm::u16vec2> tl{m_entries.try_insert(std::move(key), bitmap.size(), size(), std::forward<Args>(args)...)};
263 if (!tl.has_value()) {
264 glm::u16vec2 new_size;
265 if (m_texture.size() == glm::ivec2{}) {
266 const glm::uvec2 usize{bitmap.size()};
267 new_size = {std::bit_ceil(usize.x + 1), std::bit_ceil(usize.y + 1)};
268 tl = m_entries.try_insert(std::move(key), usize, new_size, std::forward<Args>(args)...);
269 }
270 else {
271 new_size = size();
272 do {
273 new_size.y < new_size.x ? new_size.y *= 2 : new_size.x *= 2;
274 } while (
275 !(tl = m_entries.try_insert(std::move(key), bitmap.size(), new_size, std::forward<Args>(args)...)).has_value());
276 }
277 reserve(new_size);
278 }
279 m_texture.set_region(*tl, bitmap);
280 }
