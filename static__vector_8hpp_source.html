423 template <forward_iterator_to_convertible_to<Element> Iterator, std::sentinel_for<Iterator> Sentinel>
424 requires(std::move_constructible<Element> && move_assignable<Element>)
425 iterator insert(const_iterator where, Iterator first, Sentinel last)
426 noexcept(nothrow_movable<Element> && nothrow_forward_iterator_to_convertible_to<Iterator, Element>)
427 {
428 const auto inserted_elements{std::ranges::distance(first, last)};
429
430 TR_ASSERT(static_cast<usize>(m_size + inserted_elements) <= Capacity,
431 "Tried to do an insert into a static vector that would put it past its capacity of {}.", Capacity);
432 TR_ASSERT(where >= begin() && where <= end(), "Tried to pass an invalid iterator to static_vector::insert.");
433
434 const iterator mwhere{begin() + std::ranges::distance(cbegin(), where)};
435 const difference_type affected_elements{std::ranges::distance(mwhere, end())};
436 if (affected_elements > 0) {
437 if (inserted_elements < affected_elements) {
438 std::uninitialized_move(end() - inserted_elements, end(), end());
439 std::move_backward(mwhere, end() - inserted_elements, end());
440 std::destroy_n(mwhere, inserted_elements);
441 }
442 else {
443 std::uninitialized_move(mwhere, end(), mwhere + inserted_elements);
444 std::destroy(mwhere, end());
445 }
446 }
447 std::uninitialized_copy(first, last, mwhere);
448 m_size += inserted_elements;
449 return mwhere;
450 }

484 template <typename... Args>
485 requires(std::move_constructible<Element> && move_assignable<Element> && std::constructible_from<Element, Args...>)
486 iterator emplace(const_iterator where, Args&&... args)
487 noexcept(nothrow_movable<Element> && std::is_nothrow_constructible_v<Element, Args...>)
488 {
489 TR_ASSERT(m_size < Capacity, "Tried to insert into a static vector that is already at its capacity of {}.", Capacity);
490 TR_ASSERT(where >= begin() && where <= end(), "Tried to pass an invalid iterator to static_vector::insert.");
491
492 const iterator mwhere{begin() + std::distance(cbegin(), where)};
493 if (mwhere != end()) {
494 new (end().base()) Element{std::move(back())};
495 std::move_backward(mwhere, end() - 1, end());
496 mwhere->~Element();
497 }
498 new (mwhere.base()) Element{std::forward<Args>(args)...};
499 ++m_size;
500 return mwhere;
