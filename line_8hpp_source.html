106 template <arithmetic LhsRep, arithmetic RhsRep>
107 [[nodiscard]] constexpr bool intersecting(const line_segment<LhsRep>& lhs, const line_segment<RhsRep>& rhs) noexcept
108 {
109 using BaseCommonRep = std::common_type_t<LhsRep, RhsRep>;
110 using CommonRep = std::conditional_t<std::integral<BaseCommonRep>, double, BaseCommonRep>;
111 const line_segment<CommonRep> lhs_cr{static_cast<line_segment<CommonRep>>(lhs)};
112 const line_segment<CommonRep> rhs_cr{static_cast<line_segment<CommonRep>>(rhs)};
113
114 const glm::tvec2<CommonRep> delta{rhs_cr.b - rhs_cr.a};
115 const glm::tvec2<CommonRep> normal{lhs_cr.b.y - lhs_cr.a.y, -lhs_cr.b.x + lhs_cr.a.x};
116 const CommonRep dot{glm::dot(delta, normal)};
117 if (std::abs(dot) < CommonRep{1e-6}) {
118 return false;
119 }
120 const CommonRep t{glm::dot(lhs_cr.a - rhs_cr.a, normal) / dot};
121 if (t < 0 || t > 1) {
122 return false;
123 }
124 const glm::tvec2<CommonRep> result{rhs_cr.a + delta * t};
125 const glm::tvec2<CommonRep> min{glm::min(lhs_cr.a, lhs_cr.b)};
126 const glm::tvec2<CommonRep> max{glm::max(lhs_cr.a, lhs_cr.b)};
127 return result.x >= min.x && result.x <= max.x && (result.x != lhs_cr.a.x || (result.y >= min.y && result.y <= max.y));
128 }
