160 usize keys{0};
161 bool stale_keys{false};

163
164 for (const auto& [key, value] : std::ranges::subrange{begin, end}) {
165 ++keys;
166 if (!stale_keys) {
167 if (const auto existing_copy_it{m_keys.find(key)}; existing_copy_it != m_keys.end()) {
168 stale_keys = true;
169 m_pool.erase(*existing_copy_it, std::next(*existing_copy_it, 2));
170 }
171 }
172 else {
173 for (auto it = m_pool.begin(); it != m_pool.end(); std::advance(it, 2)) {
174 if (*it == key) {
175 m_pool.erase(it, std::next(it, 2));
176 break;
177 }
178 }
179 }

181 m_pool.append(key);
182 m_pool.append(value);

184 stale_keys = true;
