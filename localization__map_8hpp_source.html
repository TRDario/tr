295 requires localization_key_value_pair<std::iter_reference_t<Iterator>>
296 void update(Iterator begin, Sentinel end)
297 {
298 try {
299 usize range_distance{0};
300 bool replaced_keys_or_reallocated_pool{false};
301 const usize old_pool_capacity{m_pool.capacity_bytes()};
302
303 for (const auto& [key, value] : std::ranges::subrange{begin, end}) {
304 ++range_distance;
305 if (!replaced_keys_or_reallocated_pool) {
306 if (const decltype(m_keys)::const_iterator key_it{m_keys.find(key)}; key_it != m_keys.end()) {
307 replaced_keys_or_reallocated_pool = true;
308 m_pool.erase(*key_it, std::ranges::next(*key_it, 2));
309 }
310 }
311 else if (const string_pool::iterator key_it{std::ranges::find(m_pool | std::views::stride(2), key).base()};
312 key_it != m_pool.end()) {
313 m_pool.erase(key_it, std::ranges::next(key_it, 2));
314 }
315
316 m_pool.append(key);
317 m_pool.append(value);
318 if (m_pool.capacity_bytes() > old_pool_capacity) {
319 replaced_keys_or_reallocated_pool = true;
320 }
321 }
322
323 if (replaced_keys_or_reallocated_pool) {
324 m_keys.clear();
325 for (string_pool::iterator it{m_pool.begin()}; it != m_pool.end(); std::ranges::advance(it, 2)) {
326 m_keys.insert(it);
327 }
328 }
329 else if (!m_pool.empty()) {
330 const string_pool::iterator rbegin{std::ranges::prev(m_pool.end(), 2)};
331 const string_pool::iterator rend{std::ranges::prev(rbegin, range_distance * 2)};
332 for (string_pool::iterator it = rbegin; it != rend; std::ranges::advance(it, -2)) {
