216 template <standard_layout Element>
217 void set_vertex_buffer(const static_vertex_buffer<Element>& buffer, int slot, int offset) noexcept
218 {
219 TR_ASSERT(buffer.valid(), "Tried to set a vertex buffer in an invalid state to a graphics context.");
220 TR_ASSERT(&buffer.context() == this, "Tried to set vertex buffer {} to a context it is not associated with.", buffer);
221
222#ifdef TR_ENABLE_CHECKED_GRAPHICS
223 check_typed_vertex_buffer(buffer.label(), slot, as_vertex_attribute_list<Element>);
224#endif
225 if (m_bound_vertex_buffers.size() <= static_cast<usize>(slot)) {
226 m_bound_vertex_buffers.resize(slot + 1);
227 }
228
229 bound_vertex_buffer_info& bound_vertex_buffer{m_bound_vertex_buffers[slot]};
230 if (internal::graphics_object_id id{buffer.id()};
231 bound_vertex_buffer.id != id || bound_vertex_buffer.offset != offset || bound_vertex_buffer.stride != sizeof(Element)) {
232 TR_LOG_TRACE("gfx", "Setting vertex buffer {} to context.", buffer);
233 gl().bind_vertex_buffer(slot, buffer.unwrap(), offset * sizeof(Element), sizeof(Element));
234 bound_vertex_buffer.id = id;
235 bound_vertex_buffer.offset = offset * sizeof(Element);
236 bound_vertex_buffer.stride = sizeof(Element);
237#ifdef TR_ENABLE_CHECKED_GRAPHICS
238 bound_vertex_buffer.vertex_attributes = {};
239#endif
240 }
241 }
