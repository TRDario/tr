222 template <standard_layout Element>
223 void set_vertex_buffer(const static_vertex_buffer<Element>& buffer, int slot, int offset) noexcept
224 {
225 TR_ASSERT(buffer.valid(), "Tried to set a vertex buffer in an invalid state to a graphics context.");
226 TR_ASSERT(&buffer.context() == this, "Tried to set vertex buffer {} to a context it is not associated with.", buffer);
227
228#ifdef TR_ENABLE_CHECKED_GRAPHICS
229 check_typed_vertex_buffer(buffer.label(), slot, as_vertex_attribute_list<Element>);
230#endif
231 if (m_bound_vertex_buffers.size() <= static_cast<usize>(slot)) {
232 m_bound_vertex_buffers.resize(slot + 1);
233 }
234
235 bound_vertex_buffer_info& bound_vertex_buffer{m_bound_vertex_buffers[slot]};
236 if (internal::graphics_object_id id{buffer.id()};
237 bound_vertex_buffer.id != id || bound_vertex_buffer.offset != offset || bound_vertex_buffer.stride != sizeof(Element)) {
238 TR_LOG_TRACE("gfx", "Setting vertex buffer {} to context.", buffer);
239 gl().bind_vertex_buffer(slot, buffer.unwrap(), offset * sizeof(Element), sizeof(Element));
240 bound_vertex_buffer.id = id;
241 bound_vertex_buffer.offset = offset * sizeof(Element);
242 bound_vertex_buffer.stride = sizeof(Element);
243#ifdef TR_ENABLE_CHECKED_GRAPHICS
244 bound_vertex_buffer.vertex_attributes = {};
245#endif
246 }
247 }
