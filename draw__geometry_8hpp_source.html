170 template <std::output_iterator<u16> Iterator>
171 constexpr Iterator fill_simple_polygon_indices(Iterator out, std::span<const glm::vec2> vertices, u16 base)
172 {
173 TR_ASSERT(vertices.size() >= 3, "Tried to calculate indices for {}-sided polygon.", vertices.size());
174 TR_ASSERT(base + vertices.size() <= UINT16_MAX, "Index overflow detected in fill_simple_polygon_indices.");
175
176 const winding_order winding_order{polygon_winding_order(vertices)};
177
178 std::vector<u16> indices(vertices.size());
179 std::iota(indices.begin(), indices.end(), 0_u16);
180 while (indices.size() >= 3) {
181 for (usize i = 0; i < indices.size(); ++i) {
182 const u16 left{static_cast<u16>(i == 0 ? indices.size() - 1 : i - 1)};
183 const u16 right{static_cast<u16>(i == indices.size() - 1 ? 0 : i + 1)};
184
185 if (indices.size() > 3) {
186 const triangle tri{vertices[indices[left]], vertices[indices[i]], vertices[indices[right]]};
187 const auto is_in_tri{[&](usize j) { return j != left && j != i && j != right && tri.contains(vertices[indices[j]]); }};
188 if (tri.winding_order() != winding_order || std::ranges::any_of(std::views::iota(0uz, indices.size()), is_in_tri)) {
189 continue;
190 }
191 }
192
193 *out++ = indices[left] + base;
194 *out++ = indices[i] + base;
195 *out++ = indices[right] + base;
196 indices.erase(indices.begin() + i--);
197 break;
198 }
199 }
200
201 return out;
202 }

411 template <std::output_iterator<glm::vec2> Iterator>
412 Iterator fill_arc_vertices(Iterator out, usize vertices, circle circle, angle start, angle size)
413 {
414 angle dth{size / vertices};
415 float dsin{dth.sin()};
416 float dcos{dth.cos()};
417 glm::vec2 delta{circle.radius * start.cos(), circle.radius * start.sin()};
418 for (usize i = 0; i < vertices; ++i) {
419 *out++ = delta + circle.center;
420 delta = glm::vec2{dcos * delta.x - dsin * delta.y, dsin * delta.x + dcos * delta.y};
421 }
422 return out;
423 }
