171 template <std::output_iterator<u16> Iterator>
172 constexpr Iterator fill_simple_polygon_indices(Iterator out, std::span<const glm::vec2> vertices, u16 base)
173 {
174 TR_ASSERT(vertices.size() >= 3, "Tried to calculate indices for {}-sided polygon.", vertices.size());
175 TR_ASSERT(base + vertices.size() <= UINT16_MAX, "Index overflow detected in fill_simple_polygon_indices.");
176
177 const winding_order winding_order{polygon_winding_order(vertices)};
178
179 std::vector<u16> indices(vertices.size());
180 std::iota(indices.begin(), indices.end(), 0_u16);
181 while (indices.size() >= 3) {
182 for (usize i = 0; i < indices.size(); ++i) {
183 const u16 left{static_cast<u16>(i == 0 ? indices.size() - 1 : i - 1)};
184 const u16 right{static_cast<u16>(i == indices.size() - 1 ? 0 : i + 1)};
185
186 if (indices.size() > 3) {
187 const triangle<float> tri{vertices[indices[left]], vertices[indices[i]], vertices[indices[right]]};
188 const auto is_in_tri{[&](usize j) { return j != left && j != i && j != right && tri.contains(vertices[indices[j]]); }};
189 if (tri.winding_order() != winding_order || std::ranges::any_of(std::views::iota(0uz, indices.size()), is_in_tri)) {
190 continue;
191 }
192 }
193
194 *out++ = indices[left] + base;
195 *out++ = indices[i] + base;
196 *out++ = indices[right] + base;
197 indices.erase(indices.begin() + i--);
198 break;
199 }
200 }
201
202 return out;
203 }

415 template <std::output_iterator<glm::vec2> Iterator>
416 Iterator fill_arc_vertices(Iterator out, usize vertices, circle<float> circle, angle<float> start, angle<float> size)
417 {
418 angle dth{size / vertices};
419 float dsin{dth.sin()};
420 float dcos{dth.cos()};
421 glm::vec2 delta{circle.radius * start.cos(), circle.radius * start.sin()};
422 for (usize i = 0; i < vertices; ++i) {
423 *out++ = delta + circle.center;
424 delta = glm::vec2{dcos * delta.x - dsin * delta.y, dsin * delta.x + dcos * delta.y};
425 }
426 return out;
427 }
