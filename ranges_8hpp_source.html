22 template <standard_layout To, standard_layout From, usize Extent>
23 [[nodiscard]] auto reinterpret_span(std::span<From, Extent> from) noexcept
24 {
25 if constexpr (Extent != std::dynamic_extent) {
26 static_assert(Extent * sizeof(From) % sizeof(To) == 0, "Cannot reinterpret span due to size_bytes() % sizeof(T) != 0.");
27 return std::span<To, Extent * sizeof(From) / sizeof(To)>{reinterpret_cast<To*>(from.data()),
28 Extent * sizeof(From) / sizeof(To)};
29 }
30 else {
31 TR_ASSERT(from.size_bytes() % sizeof(To) == 0, "Cannot reinterpret span as '{}' due to size_bytes ({}) % sizeof({}) ({}) != 0.",
32 type_name<To>(), from.size_bytes(), type_name<To>(), sizeof(To));
33 return std::span<To>{reinterpret_cast<To*>(from.data()), from.size_bytes() / sizeof(To)};
34 }

256 {
