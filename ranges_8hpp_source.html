24 [[nodiscard]] auto reinterpret_span(std::span<From, Extent> from) noexcept
25 {
26 if constexpr (Extent != std::dynamic_extent) {
27 static_assert(Extent * sizeof(From) % sizeof(To) == 0, "Cannot reinterpret span due to size_bytes() % sizeof(T) != 0.");

195 [[nodiscard]] constexpr std::ranges::borrowed_iterator_t<SearchedRange> find_last_of(SearchedRange&& searched,
196 WhitelistRange&& whitelist)
197 {
198 for (auto value_it = std::ranges::rbegin(searched); value_it != std::ranges::rend(searched); ++value_it) {
199 if (std::ranges::contains(whitelist, *value_it)) {
