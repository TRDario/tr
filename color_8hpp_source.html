1059 [[nodiscard]] constexpr hsv(rgbf rgb) noexcept
1060 {
1061 v = std::max({rgb.r, rgb.g, rgb.b});
1062 const float delta{v - std::min({rgb.r, rgb.g, rgb.b})};
1063
1064 if (delta == 0) {
1065 h = 0;
1066 }
1067 else if (v == rgb.r) {
1068 h = 60.0f * eucmod((rgb.g - rgb.b) / delta, 6.0f);
1069 }
1070 else if (v == rgb.g) {
1071 h = 60.0f * ((rgb.b - rgb.r) / delta + 2.0f);
1072 }
1073 else {
1074 h = 60.0f * ((rgb.r - rgb.g) / delta + 4.0f);
1075 }
1076
1077 s = v != 0 ? delta / v : 0;
1078 }
