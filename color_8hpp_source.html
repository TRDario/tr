1037 [[nodiscard]] constexpr hsv(rgbf rgb) noexcept
1038 {
1039 v = std::max({rgb.r, rgb.g, rgb.b});
1040 const float delta{v - std::min({rgb.r, rgb.g, rgb.b})};
1041
1042 if (delta == 0) {
1043 h = 0;
1044 }
1045 else if (v == rgb.r) {
1046 h = 60.0f * eucmod((rgb.g - rgb.b) / delta, 6.0f);
1047 }
1048 else if (v == rgb.g) {
1049 h = 60.0f * ((rgb.b - rgb.r) / delta + 2.0f);
1050 }
1051 else {
1052 h = 60.0f * ((rgb.r - rgb.g) / delta + 4.0f);
1053 }
1054
1055 s = v != 0 ? delta / v : 0;
1056 }

1100 constexpr auto to_number{

1220{

1222 const float c{hsv.v * hsv.s};

1227 r = hsv.v;

1229 b = m;

1232 r = x;
1233 g = hsv.v;
1234 b = m;
1235 }

1237 r = m;
1238 g = hsv.v;
1239 b = x;
1240 }
1241 else if (h < 240) {
1242 r = m;
1243 g = x;
1244 b = hsv.v;
1245 }
1246 else if (h < 300) {
1247 r = x;
1248 g = m;
1249 b = hsv.v;
