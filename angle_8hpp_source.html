323 [[nodiscard]] constexpr angle lerp(angle x, angle y, Ratio a) noexcept
324 {
325 x = eucmod(x, turns(1.0f));
326 y = eucmod(y, turns(1.0f));
327 if (abs(y - x) >= turns(0.5f)) {

329 }
