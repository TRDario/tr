60 requires indirectly_color_castable_to<rgba<u8>, Color>
61 [[nodiscard]] constexpr Color from_hex_code(std::string_view hex_code)
62 {
63 if (hex_code.size() != 7 && hex_code.size() != 9) {
64 throw invalid_hex_code{hex_code, std::format("must be 7 or 9 characters long, but is {}", hex_code.size())};
65 }
66 else if (hex_code[0] != '#') {
67 throw invalid_hex_code{hex_code, std::format("must start with '#', but starts with '{}'", hex_code[0])};
68 }
69 for (usize i = 1; i < hex_code.size(); ++i) {
70 const char c{hex_code[i]};
71 if (!(c >= '0' && c <= '9') && !(c >= 'a' && c <= 'f') && !(c >= 'A' && c <= 'F')) {
72 throw invalid_hex_code{hex_code, std::format("contains invalid hex character '{}' at position {}", c, i)};
73 }
74 }
75
76 constexpr auto hex_value{[](char c) -> u8 {
77 if (c >= '0' && c <= '9') {
78 return c - '0';
79 }
80 else if (c >= 'a' && c <= 'f') {
81 return c - 'a' + 10;
82 }
83 else /* c >= 'A' && c <= 'F' */ {
84 return c - 'A' + 10;
85 }
86 }};
87
88 return color_cast<Color>(rgba<u8>{
89 static_cast<u8>((hex_value(hex_code[1]) << 4) | hex_value(hex_code[2])),
90 static_cast<u8>((hex_value(hex_code[3]) << 4) | hex_value(hex_code[4])),
91 static_cast<u8>((hex_value(hex_code[5]) << 4) | hex_value(hex_code[6])),
92 static_cast<u8>(hex_code.size() == 9 ? (hex_value(hex_code[7]) << 4) | hex_value(hex_code[8]) : 255),
93 });
