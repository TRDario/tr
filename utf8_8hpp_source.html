46 template <input_iterator Iterator>
47 [[nodiscard]] constexpr codepoint to_cp(Iterator it) noexcept(noexcept(*++it))
48 {
49 const u8 leading_byte{static_cast<u8>(*it++)};
50 if (leading_byte < 0x80) {
51 return leading_byte;
52 }
53 else if (leading_byte < 0xE0) {
54 const u8 second_byte{static_cast<u8>(static_cast<u8>(*it) & 0x3F)};
55 return ((leading_byte & 0x1F) << 6) | (second_byte & 0x3F);
56 }
57 else if (leading_byte < 0xF0) {
58 const u8 second_byte{static_cast<u8>(static_cast<u8>(*it++) & 0x3F)};
59 const u8 third_byte{static_cast<u8>(static_cast<u8>(*it) & 0x3F)};
60 return ((leading_byte & 0xF) << 12) | ((second_byte & 0x3F) << 6) | (third_byte & 0x3F);
61 }
62 else {
63 const u8 second_byte{static_cast<u8>(static_cast<u8>(*it++) & 0x3F)};
64 const u8 third_byte{static_cast<u8>(static_cast<u8>(*it++) & 0x3F)};
65 const u8 fourth_byte{static_cast<u8>(static_cast<u8>(*it) & 0x3F)};
66 return ((leading_byte & 0x7) << 18) | ((second_byte & 0x3F) << 12) | ((third_byte & 0x3F) << 6) | (fourth_byte & 0x3F);
67 }
68 }

136 template <output_string String>
137 constexpr String::iterator insert(String& str, typename String::iterator where, codepoint cp)
138 {
139 if (cp < 0x80) {
140 return str.insert(where, cp);
141 }
142 else if (cp < 0x800) {
143 const std::array<char, 2> chars{
144 static_cast<char>(cp >> 6 | 0xC0),
145 static_cast<char>((cp & 0x3F) | 0x80),
146 };
147 return str.insert(where, chars.begin(), chars.end());
148 }
149 else if (cp < 0x10000) {
150 const std::array<char, 3> chars{
151 static_cast<char>(cp >> 12 | 0xE0),
152 static_cast<char>(((cp >> 6) & 0x3F) | 0x80),
153 static_cast<char>((cp & 0x3F) | 0x80),
154 };
155 return str.insert(where, chars.begin(), chars.end());
156 }
157 else {
158 const std::array<char, 4> chars{
159 static_cast<char>(cp >> 18 | 0xF0),
160 static_cast<char>(((cp >> 12) & 0x3F) | 0x80),
161 static_cast<char>(((cp >> 6) & 0x3F) | 0x80),
162 static_cast<char>((cp & 0x3F) | 0x80),
163 };
164 return str.insert(where, chars.begin(), chars.end());
165 }
