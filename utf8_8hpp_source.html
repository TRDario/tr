47 template <input_iterator Iterator>
48 [[nodiscard]] constexpr codepoint to_cp(Iterator it) noexcept(noexcept(*++it))
49 {
50 const u8 leading_byte{static_cast<u8>(*it++)};
51 if (leading_byte < 0x80) {
52 return leading_byte;
53 }
54 else if (leading_byte < 0xE0) {
55 const u8 second_byte{static_cast<u8>(static_cast<u8>(*it) & 0x3F)};
56 return ((leading_byte & 0x1F) << 6) | (second_byte & 0x3F);
57 }
58 else if (leading_byte < 0xF0) {
59 const u8 second_byte{static_cast<u8>(static_cast<u8>(*it++) & 0x3F)};
60 const u8 third_byte{static_cast<u8>(static_cast<u8>(*it) & 0x3F)};
61 return ((leading_byte & 0xF) << 12) | ((second_byte & 0x3F) << 6) | (third_byte & 0x3F);
62 }
63 else {
64 const u8 second_byte{static_cast<u8>(static_cast<u8>(*it++) & 0x3F)};
65 const u8 third_byte{static_cast<u8>(static_cast<u8>(*it++) & 0x3F)};
66 const u8 fourth_byte{static_cast<u8>(static_cast<u8>(*it) & 0x3F)};
67 return ((leading_byte & 0x7) << 18) | ((second_byte & 0x3F) << 12) | ((third_byte & 0x3F) << 6) | (fourth_byte & 0x3F);
68 }

137 template <output_string String>
138 constexpr String::iterator insert(String& str, typename String::iterator where, codepoint cp)
139 {
140 if (cp < 0x80) {
141 return str.insert(where, cp);
142 }
143 else if (cp < 0x800) {
144 const std::array<char, 2> chars{
145 static_cast<char>(cp >> 6 | 0xC0),
146 static_cast<char>((cp & 0x3F) | 0x80),
147 };
148 return str.insert(where, chars.begin(), chars.end());
149 }
150 else if (cp < 0x10000) {
151 const std::array<char, 3> chars{
152 static_cast<char>(cp >> 12 | 0xE0),
153 static_cast<char>(((cp >> 6) & 0x3F) | 0x80),
154 static_cast<char>((cp & 0x3F) | 0x80),
155 };
156 return str.insert(where, chars.begin(), chars.end());
157 }
158 else {
159 const std::array<char, 4> chars{
160 static_cast<char>(cp >> 18 | 0xF0),
161 static_cast<char>(((cp >> 12) & 0x3F) | 0x80),
162 static_cast<char>(((cp >> 6) & 0x3F) | 0x80),
163 static_cast<char>((cp & 0x3F) | 0x80),
164 };
165 return str.insert(where, chars.begin(), chars.end());
