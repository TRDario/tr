207 template <typename Error, typename Base = std::formatter<Error>>
208 struct localization_error_formatter
209 {
214 template <typename ParseContext>
215 constexpr ParseContext::iterator parse(ParseContext& context)
216 {
217 if (context.begin() != context.end() && *context.begin() != '}') {
218 throw std::format_error{"Invalid localization script error format specification."};
219 }
220 return context.begin();
221 }
222
228 template <typename FormatContext>
229 FormatContext::iterator format(const Error& error, FormatContext& context) const
230 {
231 context.advance_to(std::format_to(context.out(), "{}:", error.chunk));
232 context.advance_to(error.line != EOF ? std::format_to(context.out(), "{}: ", error) : std::format_to(context.out(), "<eof>: "));
233 return static_cast<const Base*>(this)->format_message(error, context);
234 }
235 };
236} // namespace tr

259 template <typename FormatContext>
260 FormatContext::iterator format(tr::localization_token_type type, FormatContext& context) const
261 {
262 switch (type) {
263 case tr::localization_token_type::none:
264 return std::format_to(context.out(), "none");
265 case tr::localization_token_type::symbol:
266 return std::format_to(context.out(), "symbol");
267 case tr::localization_token_type::string:

271 case tr::localization_token_type::opening_brace:
272 return std::format_to(context.out(), "'{{'");
273 case tr::localization_token_type::closing_brace:
274 return std::format_to(context.out(), "'}}'");
275 default:
276 return std::format_to(context.out(), "<unknown>");
277 }
