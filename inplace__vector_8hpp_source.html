427 template <forward_iterator_to<Element> Iterator, std::sentinel_for<Iterator> Sentinel>
428 requires std::move_constructible<Element> && std::is_move_assignable_v<Element>
429 iterator insert(const_iterator where, Iterator first, Sentinel last)
430 noexcept(std::is_nothrow_move_constructible_v<Element> && std::is_nothrow_move_assignable_v<Element> &&
431 nothrow_forward_iterator_to<Iterator, Element>)
432 {
433 const auto inserted_elements{std::ranges::distance(first, last)};
434
435 TR_ASSERT(static_cast<usize>(m_size + inserted_elements) <= Capacity,
436 "Tried to do an insert into an inplace vector that would put it past its capacity of {}.", Capacity);
437 TR_ASSERT(where >= begin() && where <= end(), "Tried to pass an invalid iterator to inplace_vector::insert.");
438
439 const iterator mwhere{begin() + std::ranges::distance(cbegin(), where)};
440 const difference_type affected_elements{std::ranges::distance(mwhere, end())};
441 if (affected_elements > 0) {
442 if (inserted_elements < affected_elements) {
443 std::uninitialized_move(end() - inserted_elements, end(), end());
444 std::move_backward(mwhere, end() - inserted_elements, end());
445 std::destroy_n(mwhere, inserted_elements);
446 }
447 else {
448 std::uninitialized_move(mwhere, end(), mwhere + inserted_elements);
449 std::destroy(mwhere, end());
450 }
451 }
452 std::uninitialized_copy(first, last, mwhere);
453 m_size += inserted_elements;
454 return mwhere;
455 }

491 template <typename... Args>
492 requires std::move_constructible<Element> && std::is_move_assignable_v<Element> && std::constructible_from<Element, Args...>
493 iterator emplace(const_iterator where, Args&&... args)
494 noexcept(std::is_nothrow_move_constructible_v<Element> && std::is_nothrow_move_assignable_v<Element> &&
495 std::is_nothrow_constructible_v<Element, Args...>)
