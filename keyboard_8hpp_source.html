56 {
57 for (usize i = 0; i < internal::scancode_names.size(); ++i) {
58 if (!internal::scancode_names[i].empty() && internal::scancode_names[i] == str) {

68 [[nodiscard]] constexpr keycode to_keycode(zstring_view str) noexcept
69 {
70 if (str.empty()) {
71 return keycode::unknown;
72 }
73
74 if (static_cast<u8>(str[0]) < 0x80) {
75 if (str[0] >= 'A' && str[0] <= 'Z') {
76 return keycode{str[0] - 'A' + 'a'};
77 }
78 return keycode{str[0]};
79 }
80 else if (str == "Backspace") {
81 return keycode{'\b'};
82 }
83 else if (str == "Tab") {
84 return keycode{'\t'};
85 }
86 else if (str == "Enter") {
87 return keycode{'\r'};
88 }
89 else if (str == "Escape") {
90 return keycode{'\x1B'};
91 }
92 else if (str == "Delete") {
93 return keycode{'\x7F'};
94 }
95
96 const scancode scan{to_scancode(str)};
97 if (std::to_underlying(scan) >= 57) {
98 return keycode{std::to_underlying(scan) | (1 << 30)};
99 }
100
101 return internal::to_keycode(str);
102 }

152 [[nodiscard]] constexpr explicit scan_chord(std::string_view str) noexcept
153 {
154 std::string_view::iterator next_delimiter_it{std::ranges::find(str, '+')};
155 while (next_delimiter_it != str.end()) {
156 std::string_view substr{str.begin(), next_delimiter_it};
157 if (substr == "Shift") {
158 mods |= keymod::shift;
159 }
160 else if (substr == "Ctrl") {
161 mods |= keymod::ctrl;
162 }
163 else if (substr == "Alt") {
164 mods |= keymod::alt;
165 }
166 else {
167 mods = keymod::unknown;
168 return;
169 }
170
171 str = std::string_view{next_delimiter_it + 1, str.end()};
172 next_delimiter_it = std::ranges::find(str, '+');
173 }
174 scan = to_scancode(str);
175 }

192 [[nodiscard]] constexpr std::string name() const
193 {
194 std::string str;

196 str.append("Ctrl+");

199 str.append("Alt+");
200 }
201 if (mods & keymod::shift) {
202 str.append("Shift+");
203 }
204 str.append(tr::name(scan));
