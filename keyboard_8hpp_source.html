56 {
57 for (usize i = 0; i < internal::scancode_names.size(); ++i) {
58 if (!internal::scancode_names[i].empty() && internal::scancode_names[i] == str) {

68 [[nodiscard]] constexpr keycode to_keycode(zstring_view str) noexcept
69 {
70 if (str.empty()) {
71 return keycode::unknown;
72 }
73
74 if (static_cast<u8>(str[0]) < 0x80) {
75 if (str[0] >= 'A' && str[0] <= 'Z') {
76 return keycode{str[0] - 'A' + 'a'};
77 }
78 return keycode{str[0]};
79 }
80 else if (str == "Backspace") {
81 return keycode{'\b'};
82 }
83 else if (str == "Tab") {
84 return keycode{'\t'};
85 }
86 else if (str == "Enter") {
87 return keycode{'\r'};
88 }
89 else if (str == "Escape") {
90 return keycode{'\x1B'};
91 }
92 else if (str == "Delete") {
93 return keycode{'\x7F'};
94 }
95
96 const scancode scan{to_scancode(str)};
97 if (std::to_underlying(scan) >= 57) {
98 return keycode{std::to_underlying(scan) | (1 << 30)};
99 }
100
101 return internal::to_keycode(str);
102 }

152 [[nodiscard]] constexpr explicit scan_chord(std::string_view str) noexcept
153 {
154 const auto first_delimiter_pos{str.find('+')};
155 if (first_delimiter_pos == str.npos) {
156 scan = to_scancode(str);
157 return;
158 }
159
160 std::string_view substr{str.begin(), str.begin() + first_delimiter_pos};
161 while (substr.end() != str.end()) {
162 if (substr == "Shift") {
163 mods |= keymod::shift;
164 }
165 else if (substr == "Ctrl") {
166 mods |= keymod::ctrl;
167 }
168 else if (substr == "Alt") {
169 mods |= keymod::alt;
170 }
171 else {
172 mods = keymod::unknown;
173 return;
174 }
175
176 const usize next_start{str.find('+', substr.end() - str.begin() + 1)};
177 substr = {substr.end() + 1, next_start == str.npos ? str.end() : str.begin() + next_start};
178 }
179 scan = to_scancode(substr);
180 }
181
185
189 [[nodiscard]] friend constexpr bool operator==(scan_chord lhs, scan_chord rhs) noexcept = default;
190
194
197 [[nodiscard]] constexpr std::string name() const
198 {
199 std::string str;
200 if (mods & keymod::ctrl) {
201 str.append("Ctrl+");
202 }
203 if (mods & keymod::alt) {
204 str.append("Alt+");
205 }
206 if (mods & keymod::shift) {
207 str.append("Shift+");
208 }
209 str.append(tr::name(scan));
210 return str;
211 }
